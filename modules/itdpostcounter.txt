(() => {
  "use strict";

  const STATE = {
    token: null,
    username: null,
    postsCount: null,
    lastRendered: null,
    lastFetchAt: 0,
    FETCH_INTERVAL: 60_000, // 1 минута
    inflight: false,

    selectors: {
      stats: ".profile-stats",
      counter: ".itd-posts-counter",
      posts: '.post-container[data-post-id]'
    }
  };

  const log = (...a) =>
    console.log("%c[ITD EXT]%c", "color:#2196F3;font-weight:700", "color:inherit", ...a);

  function getUsernameFromPath() {
    const p = location.pathname.replace(/^\/+|\/+$/g, "");
    if (!p || p.includes("/")) return null;
    if (!/^[a-zA-Z0-9_]+$/.test(p)) return null;
    return p;
  }

  async function getNewToken() {
    try {
      const res = await fetch("/api/v1/auth/refresh", {
        method: "POST",
        credentials: "include"
      });
      if (!res.ok) return null;
      const data = await res.json();
      return data.accessToken || data.token || null;
    } catch {
      return null;
    }
  }

  async function ensureToken() {
    if (STATE.token) return STATE.token;
    STATE.token = await getNewToken();
    return STATE.token;
  }

  async function fetchUser(username, forceRefreshToken = false) {
    if (forceRefreshToken) STATE.token = await getNewToken();
    const token = await ensureToken();
    if (!token) throw new Error("Нет access token (refresh не сработал)");

    const res = await fetch(`/api/users/${username}`, {
      method: "GET",
      credentials: "include",
      headers: {
        accept: "application/json, */*",
        authorization: `Bearer ${token}`
      }
    });

    if (res.status === 401 && !forceRefreshToken) {
      return fetchUser(username, true);
    }

    if (!res.ok) throw new Error(`Ошибка /api/users/${username}: ${res.status}`);
    return res.json();
  }

  function upsertCounter(postsCount) {
    const stats = document.querySelector(STATE.selectors.stats);
    if (!stats) return;

    let btn = stats.querySelector(STATE.selectors.counter);
    if (!btn) {
      btn = document.createElement("button");
      btn.type = "button";
      btn.className = "profile-stats__item svelte-p40znu itd-posts-counter";
      btn.style.cursor = "pointer";

      btn.addEventListener("click", (e) => {
        e.preventDefault();
        document.querySelector(STATE.selectors.posts)?.scrollIntoView({
          behavior: "smooth",
          block: "start"
        });
      });

      stats.appendChild(btn);
    }

    btn.innerHTML = `
      <span class="profile-stats__value svelte-p40znu">${Number.isFinite(postsCount) ? postsCount : "…"}</span>
      <span class="profile-stats__label svelte-p40znu">Посты</span>
    `;
  }

  async function refreshOnceIfNeeded() {
    const u = getUsernameFromPath();
    if (!u) return;

    if (STATE.username !== u) {
      STATE.username = u;
      STATE.lastFetchAt = 0;
      STATE.postsCount = null;
      STATE.lastRendered = null;
    }

    const now = Date.now();

    if (!document.querySelector(STATE.selectors.stats)) return;

    if (now - STATE.lastFetchAt < STATE.FETCH_INTERVAL) return;

    // защита от параллельных вызовов
    if (STATE.inflight) return;
    STATE.inflight = true;

    try {

      if (!document.querySelector(STATE.selectors.counter)) upsertCounter(null);

      STATE.lastFetchAt = now;

      const data = await fetchUser(u);
      const pc = Number.isFinite(+data?.postsCount) ? +data.postsCount : null;

      STATE.postsCount = pc;
      if (STATE.lastRendered !== pc) {
        STATE.lastRendered = pc;
        upsertCounter(pc);
        log("postsCount обновлён", { user: u, postsCount: pc });
      }
    } catch (e) {
      log("ошибка", e?.message || e);
    } finally {
      STATE.inflight = false;
    }
  }

  function hookHistory() {
    const fire = () => window.dispatchEvent(new Event("itd:navigation"));
    ["pushState", "replaceState"].forEach((m) => {
      const orig = history[m];
      history[m] = function (...args) {
        const r = orig.apply(this, args);
        fire();
        return r;
      };
    });
    window.addEventListener("popstate", fire);
  }

  function init() {
    hookHistory();

    window.addEventListener("itd:navigation", () => refreshOnceIfNeeded());

    const obs = new MutationObserver(() => refreshOnceIfNeeded());
    obs.observe(document.documentElement, { childList: true, subtree: true });

    setInterval(() => refreshOnceIfNeeded(), STATE.FETCH_INTERVAL);

    refreshOnceIfNeeded();
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", init, { once: true });
  } else {
    init();
  }
})();
